\doxysubsection{Detailed Description}
An autoloader that uses both PSR4 autoloading, and traditional classmaps.

Given a foo-\/bar package of classes in the file system at the following paths\+: 
\begin{DoxyCode}{0}
\DoxyCodeLine{/path/to/packages/foo-\/bar/}
\DoxyCodeLine{\ \ \ \ /src}
\DoxyCodeLine{\ \ \ \ \ \ \ \ Baz.php\ \ \ \ \ \ \ \ \ \#\ Foo\(\backslash\)Bar\(\backslash\)Baz}
\DoxyCodeLine{\ \ \ \ \ \ \ \ Qux/}
\DoxyCodeLine{\ \ \ \ \ \ \ \ \ \ \ \ Quux.php\ \ \ \ \#\ Foo\(\backslash\)Bar\(\backslash\)Qux\(\backslash\)Quux}

\end{DoxyCode}
 you can add the path to the configuration array that is passed in the constructor. The \doxylink{namespace_config}{Config} array consists of 2 primary keys, both of which are associative arrays\+: \textquotesingle{}psr4\textquotesingle{}, and \textquotesingle{}classmap\textquotesingle{}. 
\begin{DoxyCode}{0}
\DoxyCodeLine{\$Config\ =\ [}
\DoxyCodeLine{\ \ \ \ \textcolor{stringliteral}{'psr4'}\ =>\ [}
\DoxyCodeLine{\ \ \ \ \ \ \ \ \textcolor{stringliteral}{'Foo\(\backslash\)Bar'}\ \ \ =>\ \textcolor{stringliteral}{'/path/to/packages/foo-\/bar'}}
\DoxyCodeLine{\ \ \ \ ],}
\DoxyCodeLine{\ \ \ \ \textcolor{stringliteral}{'classmap'}\ =>\ [}
\DoxyCodeLine{\ \ \ \ \ \ \ \ \textcolor{stringliteral}{'MyClass'}\ \ \ =>\ \textcolor{stringliteral}{'/path/to/class/file.php'}}
\DoxyCodeLine{\ \ \ \ ]}
\DoxyCodeLine{];}

\end{DoxyCode}
 Example\+: 
\begin{DoxyCode}{0}
\DoxyCodeLine{<?php}
\DoxyCodeLine{\textcolor{comment}{//\ our\ configuration\ array}}
\DoxyCodeLine{\$Config\ =\ [\ ...\ ];}
\DoxyCodeLine{\$loader\ =\ new\ \(\backslash\)CodeIgniter\(\backslash\)Autoloader\(\backslash\)Autoloader(\$Config);}
\DoxyCodeLine{}
\DoxyCodeLine{\textcolor{comment}{//\ register\ the\ autoloader}}
\DoxyCodeLine{\$loader-\/>register();}

\end{DoxyCode}
